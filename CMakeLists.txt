cmake_minimum_required(VERSION 3.24)
project(UnifiedMusic VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(src/cme/Dependencies.cmake)
add_library(unified_music_core STATIC src/core/PluginScan.cpp src/core/CombatPlan.cpp)
target_include_directories(unified_music_core PUBLIC src/core)
target_link_libraries(unified_music_core PUBLIC ZLIB::ZLIB)
add_executable(UnifiedMusicInspect src/core/Inspect.cpp)
target_link_libraries(UnifiedMusicInspect PRIVATE unified_music_core)
add_subdirectory(src/cme)
add_subdirectory(src/mtd)
add_library(UnifiedMusic SHARED src/Main.cpp src/AutoCombat.cpp)
target_link_libraries(UnifiedMusic PRIVATE unified_music_core unified_music_cme unified_music_mtd)
target_compile_definitions(UnifiedMusic PRIVATE SKYRIM_AE SKYRIM_SUPPORT_AE)
if(MSVC)
    target_compile_options(UnifiedMusic PRIVATE /utf-8 /EHsc /permissive-)
    target_compile_options(unified_music_core PRIVATE /utf-8 /EHsc)
endif()
install(TARGETS UnifiedMusic RUNTIME DESTINATION SKSE/Plugins LIBRARY DESTINATION SKSE/Plugins)
