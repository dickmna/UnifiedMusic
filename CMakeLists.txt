cmake_minimum_required(VERSION 3.24)
project(UnifiedMusic VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(cmake/Dependencies.cmake)
add_library(unified_music_core STATIC src/core/PluginScan.cpp src/core/ScriptReferences.cpp src/core/Conditions.cpp src/core/CombatPlan.cpp src/core/ConditionPlan.cpp src/core/MappingPlan.cpp src/core/DistributionPlan.cpp src/core/AudioPlan.cpp src/core/AudioAssets.cpp src/core/SelectorPlan.cpp)
target_include_directories(unified_music_core PUBLIC src/core)
target_link_libraries(unified_music_core PUBLIC ZLIB::ZLIB jsoncpp_static bcrypt PRIVATE SimpleIni::SimpleIni)
target_compile_definitions(unified_music_core PRIVATE SI_NO_CONVERSION)
add_executable(UnifiedMusicInspect src/core/Inspect.cpp)
target_link_libraries(UnifiedMusicInspect PRIVATE unified_music_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(HarmoniaRuleTests tests/selector_rules.cpp)
    target_include_directories(HarmoniaRuleTests PRIVATE src/core)
    add_test(NAME composer_selector_rules COMMAND HarmoniaRuleTests)
endif()
add_library(UnifiedMusic SHARED src/Main.cpp src/AutoCombat.cpp src/AreaMusic.cpp src/NativeTracks.cpp src/AudioCoexist.cpp src/CombatSelector.cpp)
target_link_libraries(UnifiedMusic PRIVATE unified_music_core CommonLibSSE::CommonLibSSE)
target_compile_definitions(UnifiedMusic PRIVATE SKYRIM_AE SKYRIM_SUPPORT_AE)
if(MSVC)
    target_compile_options(UnifiedMusic PRIVATE /utf-8 /EHsc /permissive-)
    target_compile_options(unified_music_core PRIVATE /utf-8 /EHsc)
endif()
install(TARGETS UnifiedMusic RUNTIME DESTINATION SKSE/Plugins LIBRARY DESTINATION SKSE/Plugins)
