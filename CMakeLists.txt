cmake_minimum_required(VERSION 3.24)
project(UnifiedMusic VERSION 1.0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(HARMONIA_BUILD_INSPECTOR "Build the portable offline inspector" ON)
include(CTest)
include(cmake/Dependencies.cmake)
add_library(unified_music_core STATIC src/core/PluginScan.cpp src/core/PluginDecode.cpp src/core/ScriptReferences.cpp src/core/Conditions.cpp src/core/CombatPlan.cpp src/core/ConditionPlan.cpp src/core/MappingPlan.cpp src/core/DistributionPlan.cpp src/core/AudioPlan.cpp src/core/AudioAssets.cpp src/core/SelectorPlan.cpp)
target_include_directories(unified_music_core PUBLIC src src/core)
target_link_libraries(unified_music_core PUBLIC jsoncpp_static bcrypt PRIVATE SimpleIni::SimpleIni)
target_compile_definitions(unified_music_core PRIVATE SI_NO_CONVERSION)
if(HARMONIA_BUILD_INSPECTOR)
    add_library(unified_music_file STATIC src/offline/PluginFile.cpp)
    target_link_libraries(unified_music_file PUBLIC unified_music_core PRIVATE ZLIB::ZLIB)
endif()
if(HARMONIA_BUILD_INSPECTOR)
    add_executable(UnifiedMusicInspect src/core/Inspect.cpp)
    target_link_libraries(UnifiedMusicInspect PRIVATE unified_music_file)
endif()
if(BUILD_TESTING)
    add_executable(HarmoniaRuleTests tests/selector_rules.cpp)
    target_include_directories(HarmoniaRuleTests PRIVATE src/core)
    add_test(NAME composer_selector_rules COMMAND HarmoniaRuleTests)
    add_executable(HarmoniaReaderTests tests/native_reader.cpp)
    target_link_libraries(HarmoniaReaderTests PRIVATE unified_music_core)
    add_test(NAME native_reader_contract COMMAND HarmoniaReaderTests)
endif()
add_library(UnifiedMusic SHARED src/Main.cpp src/NativePlugin.cpp src/AutoCombat.cpp src/AreaMusic.cpp src/NativeTracks.cpp src/AudioCoexist.cpp src/CombatSelector.cpp)
target_link_libraries(UnifiedMusic PRIVATE unified_music_core CommonLibSSE::CommonLibSSE)
target_compile_definitions(UnifiedMusic PRIVATE SKYRIM_AE SKYRIM_SUPPORT_AE)
if(MSVC)
    target_compile_options(UnifiedMusic PRIVATE /utf-8 /EHsc /permissive-)
    target_compile_options(unified_music_core PRIVATE /utf-8 /EHsc)
    target_link_options(UnifiedMusic PRIVATE "/MAP:$<TARGET_FILE_DIR:UnifiedMusic>/UnifiedMusic.map")
    if(TARGET unified_music_file)
        target_compile_options(unified_music_file PRIVATE /utf-8 /EHsc)
    endif()
endif()
install(TARGETS UnifiedMusic RUNTIME DESTINATION SKSE/Plugins LIBRARY DESTINATION SKSE/Plugins)
